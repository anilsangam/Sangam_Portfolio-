/* ---------- Doors: hover cracks open; click swings it open and navigates to the page ---------- */
        (function doors() {
            const ROUTES = ['/web-developer', '/app-developer'];
            const SLUGS = ['web-developer', 'app-developer'];
            const TITLES = ['Web Development', 'App Development'];

            const sec = $('#works');
            if (!sec) return;
            const units = [...sec.querySelectorAll('.unit')];
            const ANG = 70, ANG_WALK = 70;
            let busy = false;

            const doorsState = units.map((u, i) => {
                const door = $('.door', u);
                const s = spring({
                    tension: 120,
                    friction: 14
                }, o => {
                    door.style.transform = `rotateY(${Math.max(0, o) * ANG}deg)`;
                    u.style.setProperty('--o', Math.max(0, o).toFixed(4));
                });
                return {
                    u,
                    door,
                    s,
                    way: $('.doorway', u)
                };
            });

            if ('IntersectionObserver' in window && !reduced) {
                sec.classList.add('pre');
                const io = new IntersectionObserver(es => {
                    if (es.some(e => e.isIntersecting)) {
                        sec.classList.add('in');
                        requestAnimationFrame(() => sec.classList.remove('pre'));
                        io.disconnect();
                    }
                }, { threshold: .18 });
                io.observe(sec);
            }

            const badge = $('#doorCursor');
            let over = false;
            const place = e => {
                if (badge) badge.style.transform = `translate3d(${e.clientX}px, ${e.clientY}px, 0) translate(-50%, -50%)`;
            };
            const showBadge = on => {
                over = on;
                if (badge) badge.classList.toggle('on', on);
            };

            if (canHover) {
                addEventListener('pointermove', e => {
                    if (over) place(e);
                }, { passive: true });

                units.forEach(u => {
                    const w = $('.doorway', u);
                    if (w) {
                        w.addEventListener('pointerenter', e => {
                            place(e);
                            if (!busy) showBadge(true);
                        });
                        w.addEventListener('pointerleave', () => showBadge(false));
                    }
                });
            }

            const lift = () => {
                if (!sec.classList.contains('lift')) sec.classList.add('lift');
            };
            sec.addEventListener('pointerenter', lift);

            doorsState.forEach((d, i) => {
                if (!d.way) return;
                d.way.addEventListener('focus', lift);
                if (canHover) {
                    d.way.addEventListener('mouseenter', () => {
                        if (!busy) d.s.set(.1);
                    });
                    d.way.addEventListener('mouseleave', () => {
                        if (!busy) d.s.set(0);
                    });
                }
                d.way.addEventListener('click', (e) => {
                    e.preventDefault();
                    enter(i);
                });
            });

            const INFO = [
                ['Web Development', 'End to end, from sketch to live application', ['Discovery & planning', 'Full-stack Node.js & MongoDB', 'REST APIs & responsive frontend', 'Performance & technical SEO', 'Launch & support']],
                ['App Development', 'From product scoping to app store launch', ['Product scoping & user flows', 'React Native & Expo development', 'Authentication & notifications', 'Cross-platform mobile apps', 'Deployment & updates']]
            ];
            const dinfo = $('#dinfo');
            const wallEl = $('.doors__wall', sec);
            const MOB = matchMedia('(max-width: 639.98px)');

            units.forEach((u, i) => {
                if (INFO[i]) {
                    const w = $('.doorway', u);
                    if (w) w.setAttribute('aria-description', INFO[i][1] + ': ' + INFO[i][2].join(', '));
                }
            });

            const setInfo = i => {
                if (!dinfo || !INFO[i]) return;
                const h = dinfo.querySelector('.dinfo__h');
                const t = dinfo.querySelector('.dinfo__t');
                const p = dinfo.querySelector('.dinfo__p');
                if (h) h.textContent = INFO[i][0];
                if (t) t.textContent = INFO[i][1];
                if (p) p.innerHTML = INFO[i][2].map(s => `<li>${s}</li>`).join('');
            };

            if (canHover) {
                let active = -1;
                let token = 0;
                const show = i => {
                    const my = ++token;
                    units.forEach((u, k) => u.classList.toggle('on', k === i));
                    sec.classList.add('focus');
                    const visible = units.filter(u => getComputedStyle(u).visibility !== 'hidden');
                    const vi = visible.indexOf(units[i]);
                    if (vi < 0) return;
                    const wr = wallEl.getBoundingClientRect();
                    const r = units[i].getBoundingClientRect();
                    const toRight = vi === 0 && visible[1];
                    const other = toRight ? visible[1].getBoundingClientRect() : visible[0].getBoundingClientRect();
                    const left = Math.min(r.left, other.left);
                    const right = Math.max(r.right, other.right);

                    const placeInfo = () => {
                        if (my !== token || active !== i || !dinfo) return;
                        setInfo(i);
                        const w = Math.min(right - left, 22 * 16);
                        dinfo.style.width = w + 'px';
                        dinfo.style.left = (left - wr.left + (right - left - w) / 2) + 'px';
                        dinfo.style.top = (r.top - wr.top + r.height * .5 - dinfo.offsetHeight / 2) + 'px';
                        dinfo.classList.add('on');
                    };
                    if (dinfo && dinfo.classList.contains('on')) {
                        dinfo.classList.remove('on');
                        setTimeout(placeInfo, 130);
                    } else {
                        placeInfo();
                    }
                };

                const hide = () => {
                    token++;
                    sec.classList.remove('focus');
                    units.forEach(u => u.classList.remove('on'));
                    if (dinfo) dinfo.classList.remove('on');
                };

                const set = i => {
                    if (busy) i = -1;
                    if (i === active) return;
                    active = i;
                    if (i < 0) hide();
                    else show(i);
                };

                units.forEach((u, i) => {
                    const w = $('.doorway', u);
                    if (w) {
                        w.addEventListener('pointerenter', () => set(i));
                        w.addEventListener('pointerleave', () => set(-1));
                    }
                });
            }

            function enter(i) {
                if (busy) return;
                busy = true;
                lift();
                showBadge(false);
                sec.classList.remove('focus');
                units.forEach(u => u.classList.remove('on'));
                if (dinfo) dinfo.classList.remove('on');

                const d = doorsState[i];
                if (d) {
                    doorsState.forEach((o, k) => {
                        if (k !== i) o.s.set(0);
                    });
                    d.u.classList.add('press');
                    setTimeout(() => d.u.classList.remove('press'), 260);
                    setTimeout(() => d.s.set(ANG_WALK / ANG), 140);
                }

                // Smoothly swing open and navigate directly to the destination page
                setTimeout(() => {
                    location.href = ROUTES[i];
                }, 520);
            }
        })();

        /* ---------- Instant UI Activation (No loading delay) ---------- */
        document.body.classList.add('ready');
        startScroll();
        window.dispatchEvent(new Event('intro:ready'));

        // Direct navigation to sections without '#' in URLs
        (function handleDirectNavigation() {
            const targetSection = '<%= (typeof scrollToSection !== "undefined" && scrollToSection) ? scrollToSection : "" %>';
            if (targetSection) {
                requestAnimationFrame(() => {
                    const el = document.getElementById(targetSection);
                    if (el) {
                        setTimeout(() => {
                            el.scrollIntoView({ behavior: 'smooth' });
                        }, 120);
                    }
                });
            }
            const openContact = <%= (typeof openContact !== "undefined" && openContact) ? 'true' : 'false' %>;
            if (openContact) {
                setTimeout(() => {
                    if (typeof openModal === 'function') openModal();
                }, 200);
            }
        })();

        // Click handler for section navigation in navbar (updates URL to /works or /about without '#')
        document.addEventListener('click', e => {
            const a = e.target.closest('[data-nav-section]');
            if (!a) return;
            e.preventDefault();
            const sec = a.dataset.navSection;
            const el = document.getElementById(sec);
            if (el) {
                el.scrollIntoView({ behavior: 'smooth' });
                try {
                    history.pushState(null, '', '/' + sec);
                } catch (err) {}
            }
        });

        // Click handler for Home link (scrolls to top, clean URL '/')
        document.addEventListener('click', e => {
            const a = e.target.closest('a[href="/"]');
            if (!a || location.pathname !== '/') return;
            if (e.target.closest('.backd')) return;
            e.preventDefault();
            window.scrollTo({ top: 0, behavior: 'smooth' });
            try {
                history.pushState(null, '', '/');
            } catch (err) {}
        });
